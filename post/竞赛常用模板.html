1. 质数筛

C++
const int N = 1e6 + 5;
// 质数打表
vector<int> primes;
bool is_primes[N];
void find_primers(int n){
    for(int i = 2; i <= n; i++) is_primes[i] = true;
    for(int i = 2; i <= n; i++){
        if(is_primes[i]) primes.push_back(i);
        for(int j = 0; j < primes.size() && i*primes[j] <= n; j++){
            is_primes[i * primes[j]] = false;
            if(i % primes[j] == 0) break;
        }
    }
}

2.快速幂

C++
// 快速幂
ll qpow(ll x, ll exp, ll mod){
    ll res = 1;
    x = x % mod;
    while(exp > 0){
        if(exp & 1) res = res * x % mod;
        x = x * x % mod;
        exp >>= 1;
    }
    return res;
}

3.最大公因数 && 最小公倍数 && 互质判断

C++
// 最大公因数
ll gcd(ll a, ll b){
    return b == 0 ? a : gcd(b, a%b);
}
// 最小公倍数
ll lcm(ll a, ll b){
    if(a == 0 || b == 0) return 0;

    return (a / gcd(a, b)) * b;
}
// 判断是否互质
bool is_coprime(ll a, ll b){
    return gcd(a, b) == 1;
}
