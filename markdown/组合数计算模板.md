# 组合数计算模板

```c++
#include <iostream>
#include <algorithm>

using namespace std;

typedef long long ll;

// 【配置区】根据题目要求调整这里
const ll MOD = 1000000007; // 常见的模数，逆元计算必须要求质数
const int MAXN = 100005;   // 数组大小

ll fact[MAXN];     // 阶乘数组：fact[i] = i! % MOD
ll invFact[MAXN];  // 逆元数组：invFact[i] = (i!) 的逆元 % MOD

// ==========================================================
// 函数 1: 快速幂算法 (求 base^exp % MOD)
// 作用：用于费马小定理求逆元，时间复杂度 O(log exp)
// ==========================================================
ll q_pow(ll base, ll exp) {
    ll res = 1;
    base %= MOD; // 防爆处理
    while (exp > 0) {
        // 如果当前指数最后一位是 1，说明需要把当前底数乘入结果
        if (exp % 2 == 1) res = (res * base) % MOD;
        // 底数平方，指数减半
        base = (base * base) % MOD;
        exp >>= 1;
    }
    return res;
}

// ==========================================================
// 函数 2: 核心预处理 (阶乘与逆元的线性推导)
// 作用：提前备好所有数据，时间复杂度 O(MAXN)
// ==========================================================
void init() {
    // 1. 边界条件初始化：0! = 1，0! 的逆元也是 1
    fact[0] = 1;
    invFact[0] = 1;

    // 2. 正向推导所有阶乘
    for (int i = 1; i < MAXN; i++) {
        fact[i] = (fact[i - 1] * i) % MOD;
    }

    // 3. 费马小定理求【最大阶乘】的逆元：(MAXN-1)! ^ (MOD-2) % MOD
    invFact[MAXN - 1] = q_pow(fact[MAXN - 1], MOD - 2);

    // 4. 反向线性推导所有较小阶乘的逆元 (精华优化)
    for (int i = MAXN - 2; i >= 1; i--) {
        // 公式：(i!)的逆元 = ((i+1)!)的逆元 * (i+1)
        invFact[i] = (invFact[i + 1] * (i + 1)) % MOD; 
    }
}

// ==========================================================
// 函数 3: 组合数查询
// 作用：计算 C(n, k) % MOD，时间复杂度 O(1) （n里选k个）
// ==========================================================
ll C(ll n, ll k) {
    // 特判：不合法的取法直接返回 0 (比如从 3 个苹果里挑 5 个)
    if (k < 0 || k > n) return 0;
    
    // 核心公式：n! * (k!)^-1 * ((n-k)!)^-1 % MOD
    ll num = fact[n];
    ll den = (invFact[k] * invFact[n - k]) % MOD;
    
    return (num * den) % MOD;
}

// ==========================================================
// Main 函数示例
// ==========================================================
int main() {
    // 读写速度优化 (竞赛常备)
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    // 【致命易错点】切记调用初始化函数！
    init();

    // 接下来就可以随意、极速地查询组合数了
    // 比如：
    // cout << C(5, 2) << "\n"; 
    
    return 0;
}
```
